cmake_minimum_required(VERSION 3.25)
project(DomeWalletNative VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(GNUInstallDirs)
find_package(PkgConfig REQUIRED)
find_package(nlohmann_json 3.10 REQUIRED)
pkg_check_modules(GCRYPT REQUIRED IMPORTED_TARGET libgcrypt)
pkg_check_modules(GLIB REQUIRED IMPORTED_TARGET gio-unix-2.0)
pkg_check_modules(GPGME REQUIRED IMPORTED_TARGET gpgme)
set(core ../src/runtime/kwalletbackend)
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/config-kwalletbackend.h "#define HAVE_STDINT_H 1\n")
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/kwalletbackend_export.h "#define KWALLETBACKEND_EXPORT\n")
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/kwalletbackend_debug.h "struct NativeWalletDiscardLog {template<typename T> NativeWalletDiscardLog& operator<<(const T&) {return *this;}};\n#define qCDebug(...) NativeWalletDiscardLog{}\n")
add_library(dome-wallet-core STATIC src/codec.cpp src/wallet.cpp ${core}/blockcipher.cc ${core}/blowfish.cc ${core}/cbc.cc ${core}/sha1.cc)
target_include_directories(dome-wallet-core PUBLIC include PRIVATE ${core} ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(dome-wallet-core PUBLIC PkgConfig::GCRYPT PkgConfig::GLIB PkgConfig::GPGME nlohmann_json::nlohmann_json)
target_compile_options(dome-wallet-core PRIVATE -Wall -Wextra -Wpedantic -Wno-unused-parameter)
set(daemon_sources src/main.cpp src/backend.cpp src/prompt.cpp src/bus.cpp src/wallet-api.cpp src/secrets.cpp src/transport.cpp)
set(DOME_NATIVE_PINENTRY "/usr/local/libexec/domed/pinentry-fltk" CACHE FILEPATH "Native credential helper")
if(NOT IS_ABSOLUTE "${DOME_NATIVE_PINENTRY}")
 message(FATAL_ERROR "DOME_NATIVE_PINENTRY must be absolute")
endif()
add_executable(dome-walletd ${daemon_sources})
target_link_libraries(dome-walletd PRIVATE dome-wallet-core)
target_compile_definitions(dome-walletd PRIVATE DOME_NATIVE_PINENTRY="${DOME_NATIVE_PINENTRY}")
target_compile_options(dome-walletd PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS dome-walletd RUNTIME DESTINATION ${CMAKE_INSTALL_LIBEXECDIR}/domed)
foreach(SERVICE_NAME org.kde.kwalletd5 org.kde.kwalletd6 org.kde.ksecretd org.kde.secretservicecompat org.freedesktop.secrets org.freedesktop.impl.portal.desktop.kwallet)
 configure_file(dist/service.in ${SERVICE_NAME}.service @ONLY)
 install(FILES ${CMAKE_CURRENT_BINARY_DIR}/${SERVICE_NAME}.service DESTINATION ${CMAKE_INSTALL_DATADIR}/dbus-1/services)
endforeach()
configure_file(dist/dome-wallet.service.in dome-wallet.service @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/dome-wallet.service DESTINATION lib/systemd/user)
configure_file(dist/wallet-wrapper.in wallet-wrapper @ONLY)
foreach(alias ksecretd kwalletd6)
 install(PROGRAMS ${CMAKE_CURRENT_BINARY_DIR}/wallet-wrapper DESTINATION ${CMAKE_INSTALL_BINDIR} RENAME ${alias})
endforeach()
install(FILES dist/kwallet.portal DESTINATION ${CMAKE_INSTALL_DATADIR}/xdg-desktop-portal/portals)
install(FILES README.md VALIDATION.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/kwallet-dome)
install(DIRECTORY ../LICENSES/ DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/kwallet-dome)
include(CTest)
if(BUILD_TESTING)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 add_executable(dome-wallet-testd ${daemon_sources})
 target_link_libraries(dome-wallet-testd PRIVATE dome-wallet-core)
 target_compile_definitions(dome-wallet-testd PRIVATE DOME_NATIVE_PINENTRY="${CMAKE_CURRENT_SOURCE_DIR}/tests/pinentry-fixture.py")
 target_compile_options(dome-wallet-testd PRIVATE -Wall -Wextra -Wpedantic)
 foreach(test core interop gpg)
  add_executable(native-wallet-${test} tests/${test}.cpp)
  target_link_libraries(native-wallet-${test} PRIVATE dome-wallet-core)
  target_compile_options(native-wallet-${test} PRIVATE -UNDEBUG)
 endforeach()
 add_test(NAME native-wallet-core COMMAND native-wallet-core)
 add_test(NAME native-wallet-dependencies COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/dependencies.py $<TARGET_FILE:dome-walletd>)
 set(DOME_TEST_PYTHON "" CACHE FILEPATH "Python with tests/requirements.txt installed")
 if(DOME_TEST_PYTHON)
  add_test(NAME native-wallet-service COMMAND ${DOME_TEST_PYTHON} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:dome-wallet-testd>)
  option(DOME_NATIVE_GUI_TESTS "Run actual FLTK prompts on a private Xvfb" OFF)
  if(DOME_NATIVE_GUI_TESTS)
   add_test(NAME native-wallet-fltk COMMAND ${DOME_TEST_PYTHON} ${CMAKE_CURRENT_SOURCE_DIR}/tests/fltk-prompts.py $<TARGET_FILE:dome-walletd>)
  endif()
 endif()
endif()
